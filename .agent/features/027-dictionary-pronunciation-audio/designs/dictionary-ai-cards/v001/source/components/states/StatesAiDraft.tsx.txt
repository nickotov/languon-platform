const noop = () => undefined

function suggestion(
  index: number,
  value: string,
  status: FieldSuggestion['status'] = 'available',
  stale = false,
): FieldSuggestion {
  return { id: `demo-${index}-${value}`, field: 'translation', value, status, stale, index }
}

const twoChoices = [
  suggestion(1, 'to make the most of', 'accepted'),
  suggestion(2, 'to take advantage of'),
]

const sixChoices = [
  suggestion(1, 'to make the most of', 'accepted'),
  suggestion(2, 'to take advantage of'),
  suggestion(3, 'to put to good use'),
  suggestion(4, 'to seize (an opportunity)'),
  suggestion(5, 'to capitalise on'),
  suggestion(6, 'to benefit from'),
]

const staleChoices = [
  suggestion(1, 'to make the most of', 'available', true),
  suggestion(2, 'to take advantage of', 'available', true),
]
