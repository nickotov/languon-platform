import React from 'react'
import { Loader2Icon } from 'lucide-react'

interface IconActionProps extends Omit<React.ButtonHTMLAttributes<HTMLButtonElement>, 'children'> {
  icon: React.ReactNode
  /** Accessible name, also used as the hover title. */
  label: string
  size?: 'xs' | 'sm' | 'md'
  /** Filled treatment for an active/engaged control (e.g. currently playing). */
  active?: boolean
  /** Borderless, transparent treatment for controls that sit inline with text. */
  plain?: boolean
  loading?: boolean
}

const sizes = {
  xs: 'h-7 w-7 [&_svg]:h-3.5 [&_svg]:w-3.5',
  sm: 'h-8 w-8 [&_svg]:h-4 [&_svg]:w-4',
  md: 'h-10 w-10 [&_svg]:h-[18px] [&_svg]:w-[18px]',
}

/**
 * Token-only circular icon button. Used where a control must stay small and
 * resolve its colours from the theme in both light and dark.
 * The visual size stays compact while the hit area is expanded to ~44px.
 */
export function IconAction({
  icon,
  label,
  size = 'sm',
  active = false,
  plain = false,
  loading = false,
  disabled,
  className = '',
  ...rest
}: IconActionProps) {
  const tone = active
    ? 'border-transparent bg-primary-default text-background-surface'
    : plain
      ? 'border-transparent bg-transparent text-text-secondary hover:bg-background-subtle hover:text-text-primary'
      : 'border-border-strong bg-background-surface text-text-primary hover:bg-background-subtle'

  return (
    <button
      type="button"
      aria-label={label}
      title={label}
      aria-busy={loading || undefined}
      disabled={disabled || loading}
      className={`relative inline-flex shrink-0 items-center justify-center rounded-full border transition-colors before:absolute before:-inset-2 before:content-[''] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-border-focus focus-visible:ring-offset-2 focus-visible:ring-offset-background-surface disabled:cursor-not-allowed disabled:opacity-50 ${tone} ${sizes[size]} ${className}`}
      {...rest}
    >
      {loading ? <Loader2Icon aria-hidden className="animate-spin" /> : icon}
    </button>
  )
}
