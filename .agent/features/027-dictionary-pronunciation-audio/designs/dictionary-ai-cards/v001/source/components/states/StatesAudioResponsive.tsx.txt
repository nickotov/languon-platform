import React from 'react'
import { StateFrame, StateSection } from './StateFrame'
import { Card } from '../Card'
import { Button } from '../Button'
import { Input } from '../Input'
import { Badge } from '../Badge'
import { InlineAlert } from '../InlineAlert'
import { FieldAudioControl } from '../audio/FieldAudioControl'
import { CardRow } from '../workspace/CardRow'
import type { AudioPhase, AudioSelection } from '../../types/audio'
import { dictionaries } from '../../data/dictionaries'
import { cards } from '../../data/cards'

const noop = () => undefined
const spanish = dictionaries[0]
const arabic = dictionaries[1]
const aprovechar = cards[0]
const arabicCard = cards[6]

function selectionFor(phase: AudioPhase, extra: Partial<AudioSelection> = {}): AudioSelection {
  return {
    cardId: 'demo-card',
    field: 'source',
    phase,
    speed: 1,
    developmentTone: false,
    cached: true,
    ...extra,
  }
}

function AudioRow({
  label,
  selection,
  availability = 'enabled',
  cached = true,
  developmentAudio = false,
  language = 'es',
  behaviour = 'normal',
}: {
  label: string
  selection: AudioSelection | null
  availability?: 'enabled' | 'cached-only' | 'disabled'
  cached?: boolean
  developmentAudio?: boolean
  language?: 'es' | 'en' | 'he'
  behaviour?: 'normal' | 'blocked-autoplay' | 'failure' | 'unavailable'
}) {
  return (
    <div className="flex flex-wrap items-center gap-3 border-b border-border-default py-2 last:border-b-0">
      <span className="w-40 shrink-0 text-xs font-medium text-text-secondary">{label}</span>
      <FieldAudioControl
        cardId="demo-card"
        field="source"
        fieldLabel="source word"
        language={language}
        availability={availability}
        cached={cached}
        developmentAudio={developmentAudio}
        behaviour={behaviour}
        selection={selection}
        onPlay={noop}
        onStop={noop}
        onSpeed={noop}
      />
    </div>
  )
}
