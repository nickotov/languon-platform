import React, { useState } from 'react'
import { CheckIcon, RefreshCwIcon, SparklesIcon, XIcon, ClockIcon } from 'lucide-react'
import { Badge } from '../Badge'
import { Button } from '../Button'
import type { FieldSuggestion } from '../../types/ai'
import type { LanguageCode, TextDirection } from '../../types/dictionary'
import { MAX_SUGGESTIONS_PER_FIELD } from '../../hooks/useCardDraft'

interface SuggestionListProps {
  fieldLabel: string
  suggestions: FieldSuggestion[]
  language: LanguageCode
  dir: TextDirection
  stale: boolean
  busy: boolean
  onAccept: (id: string) => void
  onDiscard: (id: string) => void
  onRegenerate: () => void
  regenerateDisabledReason?: string
}

export function SuggestionList({
  fieldLabel,
  suggestions,
  language,
  dir,
  stale,
  busy,
  onAccept,
  onDiscard,
  onRegenerate,
  regenerateDisabledReason,
}: SuggestionListProps) {
  const [expanded, setExpanded] = useState(false)
  if (suggestions.length === 0) return null

  const visible = expanded ? suggestions : suggestions.slice(0, 2)
  const hidden = suggestions.length - visible.length
  const atLimit = suggestions.length >= MAX_SUGGESTIONS_PER_FIELD

  return (
    <section
      aria-label={`AI suggestions for ${fieldLabel}`}
      className="mt-2 rounded-xl border border-dashed border-border-strong bg-background-subtle p-3"
    >
      <div className="flex flex-wrap items-center justify-between gap-2">
        <p className="inline-flex items-center gap-1.5 text-xs font-medium uppercase tracking-wide text-text-secondary">
          <SparklesIcon aria-hidden className="h-3.5 w-3.5" />
          Suggested · not saved
          <span className="font-normal normal-case tracking-normal text-text-tertiary">
            {suggestions.length} of {MAX_SUGGESTIONS_PER_FIELD}
          </span>
        </p>
        <Button
          size="compact"
          variant="secondary"
          leadingIcon={<RefreshCwIcon aria-hidden className="h-4 w-4" />}
          onClick={onRegenerate}
          disabled={busy || stale || atLimit || Boolean(regenerateDisabledReason)}
        >
          Regenerate {fieldLabel.toLowerCase()}
        </Button>
      </div>

      {atLimit ? (
        <p className="mt-2 text-xs text-text-secondary">
          Six choices are shown — discard one before generating another.
        </p>
      ) : null}
      {regenerateDisabledReason ? (
        <p className="mt-2 text-xs text-text-secondary">{regenerateDisabledReason}</p>
      ) : null}

      <ul className="mt-2 space-y-2">
        {visible.map((suggestion) => {
          const isAccepted = suggestion.status === 'accepted'
          const isStale = stale || suggestion.stale
          return (
            <li
              key={suggestion.id}
              className={`rounded-lg border bg-background-surface p-2.5 ${
                isAccepted
                  ? 'border-success-default'
                  : isStale
                    ? 'border-border-default opacity-80'
                    : 'border-border-default'
              }`}
            >
              <div className="flex flex-wrap items-start justify-between gap-2">
                <div className="min-w-0 flex-1">
                  <p className="text-xs font-medium text-text-tertiary">
                    Choice {suggestion.index}
                  </p>
                  <p
                    dir={dir}
                    lang={language}
                    className="mt-0.5 break-words text-sm italic text-text-primary"
                  >
                    {suggestion.value}
                  </p>
                  <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
                    {isAccepted ? (
                      <Badge
                        size="sm"
                        tone="success"
                        icon={<CheckIcon aria-hidden className="h-3.5 w-3.5" />}
                      >
                        Accepted into the field
                      </Badge>
                    ) : null}
                    {isStale ? (
                      <Badge
                        size="sm"
                        tone="warning"
                        icon={<ClockIcon aria-hidden className="h-3.5 w-3.5" />}
                      >
                        Stale — Source changed
                      </Badge>
                    ) : null}
                  </div>
                </div>
                <div className="flex shrink-0 flex-wrap gap-1">
                  <Button
                    size="compact"
                    variant={isAccepted ? 'secondary' : 'primary'}
                    leadingIcon={<CheckIcon aria-hidden className="h-4 w-4" />}
                    onClick={() => onAccept(suggestion.id)}
                    disabled={isStale || isAccepted}
                  >
                    {isAccepted ? 'Accepted' : 'Accept'}
                  </Button>
                  <Button
                    size="compact"
                    variant="secondary"
                    leadingIcon={<XIcon aria-hidden className="h-4 w-4" />}
                    onClick={() => onDiscard(suggestion.id)}
                  >
                    Discard
                  </Button>
                </div>
              </div>
            </li>
          )
        })}
      </ul>

      {hidden > 0 ? (
        <Button
          size="compact"
          variant="secondary"
          className="mt-2"
          aria-expanded={expanded}
          onClick={() => setExpanded(true)}
        >
          Show {hidden} more {hidden === 1 ? 'choice' : 'choices'}
        </Button>
      ) : null}
      {expanded && suggestions.length > 2 ? (
        <Button
          size="compact"
          variant="secondary"
          className="mt-2"
          aria-expanded={expanded}
          onClick={() => setExpanded(false)}
        >
          Show fewer choices
        </Button>
      ) : null}
    </section>
  )
}
