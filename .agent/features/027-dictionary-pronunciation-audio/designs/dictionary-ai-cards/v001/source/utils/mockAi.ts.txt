const generic: Record<CardFieldKey, string[]> = {
  source: [],
  translation: [
    'suggested translation',

    'a more idiomatic translation',
    'a formal-register translation',

  exampleTranslation: [
    'The example sentence rendered in the opposite language.',
    'A closer, more literal rendering of the example sentence.',
    'A natural-sounding rendering of the example sentence.',
  ],
}

export function suggestionFor(
  field: CardFieldKey,
  source: string,
  attempt: number,
): string {
  const key = source.trim().toLowerCase()
  const pool = pools[key]?.[field] ?? generic[field]
  if (pool.length === 0) return source
  const base = pool[attempt % pool.length]
  const cycle = Math.floor(attempt / pool.length)
  return cycle === 0 ? base : `${base} (variant ${cycle + 1})`
}

export function reviewAlternatives(
  field: CardFieldKey,
  source: string,
  count = 3,
): string[] {
  return Array.from({ length: count }, (_unused, index) =>
    suggestionFor(field, source, index + 1),
  )
}
