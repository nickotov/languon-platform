import React from 'react'

export type CardVariant = 'elevated' | 'outlined' | 'subtle'
export type CardPadding = 'none' | 'sm' | 'md' | 'lg'

export interface CardProps extends React.HTMLAttributes<HTMLDivElement> {
  /** Visual treatment of the container surface. */
  variant?: CardVariant
  /** Inner spacing applied to the card body. */
  padding?: CardPadding
  /** Adds hover/focus affordances for clickable cards. */
  interactive?: boolean
  /** Renders the card as a different element (e.g. 'article', 'section'). */
  as?: 'div' | 'article' | 'section' | 'li'
  children?: React.ReactNode
}

const variantClasses: Record<CardVariant, string> = {
  elevated: 'bg-white border border-slate-200 shadow-sm',
  outlined: 'bg-white border border-slate-300',
  subtle: 'bg-slate-50 border border-slate-100',
}

const paddingClasses: Record<CardPadding, string> = {
  none: 'p-0',
  sm: 'p-4',
  md: 'p-6',
  lg: 'p-8',
}

export function Card({
  variant = 'elevated',
  padding = 'md',
  interactive = false,
  as = 'div',
  className = '',
  children,
  ...rest
}: CardProps) {
  const Component = as as React.ElementType

  const interactiveClasses = interactive
    ? 'cursor-pointer transition-colors transition-shadow duration-150 hover:border-slate-300 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-slate-900 focus-visible:ring-offset-2'
    : ''

  return (
    <Component
      className={[
        'rounded-2xl text-slate-900',
        variantClasses[variant],
        paddingClasses[padding],
        interactiveClasses,
        className,
      ]
        .filter(Boolean)
        .join(' ')}
      {...(interactive && !rest.tabIndex ? { tabIndex: 0 } : {})}
      {...rest}
    >
      {children}
    </Component>
  )
}

export interface CardSectionProps extends React.HTMLAttributes<HTMLDivElement> {
  children?: React.ReactNode
}

export function CardHeader({ className = '', children, ...rest }: CardSectionProps) {
  return (
    <div className={['mb-4 flex flex-col gap-1', className].filter(Boolean).join(' ')} {...rest}>
      {children}
    </div>
  )
}

export function CardTitle({
  className = '',
  children,
  ...rest
}: React.HTMLAttributes<HTMLHeadingElement>) {
  return (
    <h3
      className={['text-base font-semibold leading-6 text-slate-900', className]
        .filter(Boolean)
        .join(' ')}
      {...rest}
    >
      {children}
    </h3>
  )
}

export function CardDescription({
  className = '',
  children,
  ...rest
}: React.HTMLAttributes<HTMLParagraphElement>) {
  return (
    <p className={['text-sm leading-5 text-slate-500', className].filter(Boolean).join(' ')} {...rest}>
      {children}
    </p>
  )
}

export function CardContent({ className = '', children, ...rest }: CardSectionProps) {
  return (
    <div className={['text-sm leading-6 text-slate-600', className].filter(Boolean).join(' ')} {...rest}>
      {children}
    </div>
  )
}

export function CardFooter({ className = '', children, ...rest }: CardSectionProps) {
  return (
    <div
      className={['mt-6 flex items-center gap-3 border-t border-slate-100 pt-4', className]
        .filter(Boolean)
        .join(' ')}
      {...rest}
    >
      {children}
    </div>
  )
}
